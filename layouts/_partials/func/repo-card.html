{{- /* A gh-card.dev repository card URL (parsed) -> a local SVG card built from data/projects.toml. Returns dict url/width/height, or false when the repo is not in the data file. */ -}} {{- $out := false -}} {{- if eq .Host "gh-card.dev" -}} {{- with findRESubmatch `^/repos/([^/]+)/([^/]+)\.svg$` .Path 1 -}} {{- $owner := index (index . 0) 1 -}} {{- $repo := index (index . 0) 2 -}} {{- range hugo.Data.projects.projects -}} {{- if eq (lower .name) (lower $repo) -}} {{- $lines := slice -}} {{- $cur := "" -}} {{- range split .description " " -}} {{- if and $cur (gt (add (len $cur) (len .) 1) 54) -}} {{- $lines = $lines | append $cur -}}{{- $cur = . -}} {{- else -}} {{- $cur = trim (printf "%s %s" $cur .) " " -}} {{- end -}} {{- end -}} {{- $lines = $lines | append $cur -}} {{- $foot := add 64 (mul 19 (len $lines)) -}} {{- $h := add $foot 22 -}} {{- /* Rosé Pine colours; an cannot load the site's web fonts, so system fonts back the stacks. */ -}} {{- $mono := "JetBrains Mono, ui-monospace, SFMono-Regular, Menlo, Consolas, monospace" -}} {{- $sans := "Instrument Sans, system-ui, -apple-system, Segoe UI, Roboto, Arial, sans-serif" -}} {{- $svg := printf `` $svg (add $foot 6) $mono (htmlEscape $meta) -}} {{- $res := resources.FromString (printf "cards/%s.svg" (lower .name)) $svg -}} {{- $out = dict "url" $res.RelPermalink "width" 440 "height" $h -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
